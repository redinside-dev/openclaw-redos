## Executive Summary

**Critical Finding:** 43+ sessions_send timeouts in last 3 days (40-50% timeout rate) due to embedded run timeouts (600s/10min limit). This is blocking autonomous operation and causing SLA breaches.

**Root Cause:** Embedded run timeouts - agents taking too long to respond to sessions_send calls. Complex tasks exceed 10-minute processing window.

**Impact:** Agents can't coordinate effectively, work gets duplicated/dropped, SLA breaches cascade.

## Identified Timeout Scenarios

### 1. Escalation Timeouts (P1 Issues)
**Pattern:** Agent tries to escalate critical issue, sessions_send times out
**Example:** `main → eng: P1 escalation: TICKET-20260301-039`
**Impact:** Critical issues don't get escalated, SLA breaches worsen

### 2. Status Check Timeouts
**Pattern:** Agent checks status of another agent, no response
**Example:** ENG checking on RESEARCH status
**Impact:** Coordination breaks down, duplicate work

### 3. Handoff Timeouts
**Pattern:** Agent tries to hand off work, receiving agent doesn't respond
**Example:** RESEARCH → ENG brief delivery
**Impact:** Work gets stuck, productivity drops

### 4. Context Overload Timeouts
**Pattern:** Large context in sessions_send message causes processing delays
**Example:** Sending full research brief in message body
**Impact:** Token limits causing delays, timeouts

### 5. Agent Unavailability Timeouts
**Pattern:** Receiving agent not running inner loop or busy with other task
**Example:** Agent crashed/stuck or processing long-running task
**Impact:** Work handoffs drop, productivity drops

```javascript
// Enhanced sessions_send with retry logic
async function sendWithRetry(agent, message, maxRetries = 3) {
  const handoffId = `handoff-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
  const handoffPath = `/Users/redinside/.openclaw/workspace/handoffs/${handoffId}.json`;
  
  // Write context to file before attempting
  const context = {
    handoff_id: handoffId,
    from_agent: 'allrounder',
    to_agent: agent,
    message: message,
    retry_count: 0,
    max_retries: maxRetries,
    timestamp: new Date().toISOString(),
    status: 'pending'
  };
  
  await writeFileSync(handoffPath, JSON.stringify(context, null, 2));
  
  for (let i = 0; i < maxRetries; i++) {
    const timeout = 60000 * Math.pow(2, i); // 60s, 120s, 180s
    
    try {
      console.log(`Attempt ${i + 1}/${maxRetries} to ${agent} (timeout: ${timeout}s)`);
      
      // Use sessions_send with increased timeout
      const result = await sessions_send({
        agentId: agent,
        message: `Handoff: ${handoffId}\n${message}`,
        timeoutSeconds: Math.floor(timeout / 1000)
      });
      
      // Update context with success
      context.status = 'success';
      context.retry_count = i + 1;
      context.completed_at = new Date().toISOString();
      await writeFileSync(handoffPath, JSON.stringify(context, null, 2));
      
      console.log(`Successfully sent to ${agent} after ${i + 1} attempt(s)`);
      return result;
      
    } catch (err) {
      console.log(`Attempt ${i + 1} failed: ${err.message}`);
      
      // Update context with failure
      context.retry_count = i + 1;
      context.last_error = err.message;
      await writeFileSync(handoffPath, JSON.stringify(context, null, 2));
      
      if (i === maxRetries - 1) {
        // Final attempt failed - log failure and fallback
        console.log(`All ${maxRetries} attempts failed to ${agent}`);
        
        // Log to failures directory
        const failurePath = `/Users/redinside/.openclaw/workspace/handoffs/failures/${handoffId}.json`;
        await writeFileSync(failurePath, JSON.stringify(context, null, 2));
        
        // Fallback to Telegram
        try {
          console.log(`Falling back to Telegram for ${agent}`);
          await message.send({
            channel: 'telegram',
            target: `@${agent}`,
            message: `URGENT: Failed to reach ${agent}\n\n${message}\n\nContext: ${handoffId}`,
            asVoice: false
          });
          
          // Update context with fallback success
          context.status = 'fallback_success';
          context.fallback_method = 'telegram';
          await writeFileSync(failurePath, JSON.stringify(context, null, 2));
          
          return { fallback: true, handoffId };
          
        } catch (telegramErr) {
          console.log(`Telegram fallback also failed: ${telegramErr.message}`);
          
          // Update context with complete failure
          context.status = 'complete_failure';
          context.fallback_method = 'telegram';
          context.fallback_error = telegramErr.message;
          await writeFileSync(failurePath, JSON.stringify(context, null, 2));
          
          throw new Error(`Failed to reach ${agent} after ${maxRetries} retries and Telegram fallback: ${telegramErr.message}`);
        }
      }
      
      // Wait before next retry (exponential backoff)
      await sleep(5000 * Math.pow(2, i)); // 5s, 10s, 20s
    }
  }
}

function sleep(ms) {
  return new Promise(resolve => setTimeout(resolve, ms));
}
```

#### 1.3 Add Circuit Breaker Pattern
```javascript
const circuitBreaker = {
  failures: {},
  threshold: 3,
  timeout: 300000, // 5 minutes
  
  recordFailure(agent) {
    if (!this.failures[agent]) {
      this.failures[agent] = {
        count: 0,
        lastFailure: Date.now(),
        open: false
      };
    }
    
    const record = this.failures[agent];
    record.count++;
    record.lastFailure = Date.now();
    
    if (record.count >= this.threshold) {
      record.open = true;
      record.openTime = Date.now();
      console.log(`Circuit breaker opened for ${agent} after ${record.count} failures`);
    }
  },
  
  isOpen(agent) {
    const record = this.failures[agent];
    if (!record) return false;
    
    if (record.open) {
      // Check if timeout has passed
      if (Date.now() - record.openTime > this.timeout) {
        // Reset circuit breaker
        record.open = false;
        record.count = 0;
        console.log(`Circuit breaker reset for ${agent}`);
        return false;
      }
      return true;
    }
    return false;
  },
  
  reset(agent) {
    if (this.failures[agent]) {
      this.failures[agent].count = 0;
      this.failures[agent].open = false;
    }
  }
};
```

```javascript
async function checkSLA(handoffId, priority) {
  const ackPath = `/Users/redinside/.openclaw/workspace/handoffs/acknowledgments/${handoffId}.json`;

    case 'P2': return 86400000; // 24 hours
```

```javascript
// Queue-based handoff pattern
class HandoffQueue {
  constructor() {
    this.queue = [];
    this.processing = new Set();
  }
  
  async addHandoff(agent, message) {
    const handoffId = `queue-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
    const handoff = {
      handoff_id: handoffId,
      agent: agent,
      message: message,
      timestamp: new Date().toISOString(),
      status: 'queued'
    };
    
    this.queue.push(handoff);
    console.log(`Handoff queued: ${handoffId} for ${agent}`);
    
    return handoffId;
  }
  
  async processQueue() {
    if (this.processing.size > 0) {
      console.log('Queue already processing');
      return;
    }
    
    while (this.queue.length > 0) {
      const handoff = this.queue.shift();
      this.processing.add(handoff.handoff_id);
      
      try {
        console.log(`Processing handoff: ${handoff.handoff_id} for ${handoff.agent}`);
        
        // Use our enhanced sendWithRetry
        const result = await sendWithRetry(handoff.agent, handoff.message, 3);
        
        console.log(`Completed handoff: ${handoff.handoff_id}`);
        
      } catch (err) {
        console.log(`Failed to process handoff: ${handoff.handoff_id} - ${err.message}`);
      } finally {
        this.processing.delete(handoff.handoff_id);
      }
    }
  }
  
  getQueueSize() {
    return this.queue.length;
  }
}
```

```javascript
class AgentHealthMonitor {
  constructor() {
    this.agents = {};
    this.checkInterval = 60000; // 1 minute
  }
  
  async monitorAgent(agent) {
    if (!this.agents[agent]) {
      this.agents[agent] = {
        lastResponse: Date.now(),
        responseTimes: [],
        failureCount: 0,
        healthy: true
      };
    }
    
    const agentData = this.agents[agent];
    const now = Date.now();
    
    // Check if agent has responded recently
    if (now - agentData.lastResponse > 300000) { // 5 minutes
      agentData.healthy = false;
      console.log(`Agent ${agent} may be unresponsive`);
      
      // Trigger circuit breaker
      circuitBreaker.recordFailure(agent);
    }
  }
  
  recordResponse(agent, responseTime) {
    const agentData = this.agents[agent];
    if (agentData) {
      agentData.lastResponse = Date.now();
      agentData.responseTimes.push(responseTime);
      agentData.healthy = true;
      
      // Keep only last 10 response times
      if (agentData.responseTimes.length > 10) {
        agentData.responseTimes.shift();
      }
    }
  }
  
  getAgentStatus(agent) {
    const agentData = this.agents[agent];
    if (!agentData) return { healthy: false };
    
    const avgResponseTime = agentData.responseTimes.reduce((a, b) => a + b, 0) / agentData.responseTimes.length;
    
    return {
      healthy: agentData.healthy,
      avgResponseTime: avgResponseTime,
      failureCount: agentData.failureCount,
      lastResponse: new Date(agentData.lastResponse).toISOString()
    };
  }
}
```
